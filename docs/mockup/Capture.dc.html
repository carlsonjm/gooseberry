import QtQuick
import org.kde.kadunce.spread

Item {
    id: gesture

    // Distance the finger travels before Spread commits.
    property real commitFraction: 0.5
    property real flickVelocity: 1400   // px/s
    property bool following: false

    signal committed()
    signal cancelled()

    DragHandler {
        id: drag
        target: null
        yAxis.enabled: true
        onActiveChanged: {
            if (active) { gesture.following = true; return }
            const v = drag.centroid.velocity.y
            if (-v > gesture.flickVelocity
                    || progress > gesture.commitFraction)
                gesture.committed()
            else
                gesture.cancelled()
        }
    }
}
